combat:技能名稱                      命中  躲閃  招架  傷害  內功  難度  級別  門派
       無影穿心掌(wuying-zhang)      140    0    140   140    0    100   超級  唐門   
備註： 唐門就一個終極，其餘高級，多一個超級不過分麼，還有命中方面算法稍微高點，因為不是終極技能
，而且現在都是打塔用，命中不高出來了也是一點用都沒的。



#include <ansi.h>
#include <combat.h>

inherit F_SSERVER;

string name() { return HIR "散花落葉" NOR; }

int perform(object me)
{
        string msg;
        object weapon, target;
        int skill, ap, dp;

        if (! target) target = offensive_target(me);

        if (! target || ! me->is_fighting(target))
                return notify_fail(name() + "只能對戰鬥中的對手使用。\n");

        if( query_temp("weapon", me) || query_temp("secondary_weapon", me) )
                return notify_fail(name() + "只能空手使用。\n");

        if( !objectp(weapon=query_temp("weapon", target)) )
                return notify_fail("對方沒有使用兵器，難以施展" + name() + "。\n");

        if ((int)me->query_skill("force") < 260)
                return notify_fail("你內功修為不夠，難以施展" + name() + "。\n");

        if( query("max_neili", me)<3000 )
                return notify_fail("你內力修為不夠，難以施展" + name() + "。\n");

        if ((int)me->query_skill("wuying-zhang", 1) < 180)
                return notify_fail("你無影穿心掌火候不夠，難以施展" + name() + "。\n");

        if (me->query_skill_mapped("strike") != "wuying-zhang")
                return notify_fail("你沒有激發無影穿心掌，難以施展" + name() + "。\n");

        if( query("neili", me)<400 )
                return notify_fail("你現在真氣不夠，難以施展" + name() + "。\n");

        if (! living(target))
                return notify_fail("對方都已經這樣了，用不著這麼費力吧？\n");

        ap = attack_power(me, "strike") + attack_power(me, "force-cognize");
        dp = defense_power(target, "force") + defense_power(target, "force-cognize");
        if( objectp(weapon=query_temp("weapon", target)) )
                msg = HBYEL "$N" HBYEL "雙手翻轉，如蝶入花叢，翩然飛舞，同時目射奇芒，攝人心魄。左手卻在對方目眩之際，劃出一道曼妙的弧線，輕拍在"weapon->name()"上,一股內勁驟然湧出，襲向"$n"周身經脈。
        if (ap * 3 / 2) > dp)                {
                        addn("neili", -300, me);
                       
                msg += HIR "$n" HIR "心中大凜，連忙運起內力相抗，卻為時已晚，虎口劇痛，手中" + weapon->name() 已無法掌控，脫手而出，同時身軀巨震，被一股沛然渾厚的內勁侵入氣海，驚懼下運功抵擋，卻已無法行動半分。
                me->start_busy(1);      
                target->start_busy(ap / 30 + 1);
                weapon->move(environment(me));

        對方手中沒有兵器:
                msg = HBYEL "$N" HBYEL "雙手翻轉，如蝶入花叢，翩然飛舞，同時目射奇芒，攝人心魄。左手卻在對方目眩之際，劃出一道曼妙的弧線，輕拍在"$n"身上,一股內勁驟然湧出，襲向"$n"周身經脈。
        if (ap * 3 / 2) > dp)                {
                        addn("neili", -300, me);
                       
        msg += HIR "$n" HIR "心中大凜，連忙運起內力相抗，卻為時已晚，身軀巨震，被一股沛然渾厚的內勁侵入氣海，驚懼下運功抵擋，卻已無法行動半分。
                me->start_busy(1);      
        target->start_busy(ap / 30 + 1);

        else                      addn("neili", -200, me);
        msg += CYN "$n" CYN "不慌不忙，內功瞬間運布全身，神智已復，氣勁到處，已然將$N" CYN "的左掌震開，輕易的化解了此招。
        me->start_busy(3);
                addn("neili", -300, me);
        target->start_busy(1);
